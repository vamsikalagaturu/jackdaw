# <symbol>[] and <symbol>_size, NUL-terminated
function(jd_embed out symbol file)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${file})
  file(READ ${file} hex HEX)
  string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," bytes "${hex}")
  file(WRITE ${out}.tmp "const unsigned char ${symbol}[] = {${bytes}0x00};\nconst int ${symbol}_size = sizeof(${symbol}) - 1;\n")
  configure_file(${out}.tmp ${out} COPYONLY)
endfunction()

# Lucide icons, looked up by file name through jd_icon_files (src/ui/icons.c)
file(GLOB JD_ICONS CONFIGURE_DEPENDS ${JD_ASSETS}/icons/*.svg)
set(JD_ICON_DECLS "")
set(JD_ICON_TABLE "")
foreach(icon ${JD_ICONS})
  get_filename_component(name ${icon} NAME_WE)
  string(MAKE_C_IDENTIFIER ${name} id)
  jd_embed(${JD_GEN}/icon_${id}.c jd_icon_${id} ${icon})
  list(APPEND JD_ASSET_SOURCES ${JD_GEN}/icon_${id}.c)
  string(APPEND JD_ICON_DECLS "extern const unsigned char jd_icon_${id}[];\nextern const int jd_icon_${id}_size;\n")
  string(APPEND JD_ICON_TABLE "  {\"${name}\", jd_icon_${id}, &jd_icon_${id}_size},\n")
endforeach()
file(WRITE ${JD_GEN}/icon_table.c.tmp "#include \"ui/icons.h\"\n${JD_ICON_DECLS}const jd_icon_file jd_icon_files[] = {\n${JD_ICON_TABLE}  {0, 0, 0}\n};\n")
configure_file(${JD_GEN}/icon_table.c.tmp ${JD_GEN}/icon_table.c COPYONLY)
list(APPEND JD_ASSET_SOURCES ${JD_GEN}/icon_table.c)

# Two separate things, fixed at configure time. JD_RELEASE=1 in the environment builds the app "jackdaw"; without
# it, "jackdaw-dev", with its own settings, mail, keyring entries and runtime files, so it never touches an installed
# release. The version is plain X.Y.Z only for a checkout that is exactly tag vX.Y.Z with no changes (what
# platform/release.sh builds); any other build is X.Y.Z-dev+hash[.dirty], even an installed "jackdaw".
set(JD_APP_NAME jackdaw)
set(JD_APP_ID io.github.vamsikalagaturu.jackdaw)
if(NOT "$ENV{JD_RELEASE}" STREQUAL "1")
  set(JD_APP_NAME jackdaw-dev)
  set(JD_APP_ID io.github.vamsikalagaturu.jackdaw.dev)
endif()
set(JD_VERSION_STRING "${PROJECT_VERSION}-dev")
find_package(Git QUIET)
if(GIT_FOUND)
  execute_process(COMMAND "${GIT_EXECUTABLE}" rev-parse --short=7 HEAD WORKING_DIRECTORY "${JD_REPO}"
                  OUTPUT_VARIABLE JD_GIT_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET RESULT_VARIABLE JD_GIT_FAILED)
  execute_process(COMMAND "${GIT_EXECUTABLE}" status --porcelain --untracked-files=no
                  WORKING_DIRECTORY "${JD_REPO}" OUTPUT_VARIABLE JD_GIT_CHANGES ERROR_QUIET)
  execute_process(COMMAND "${GIT_EXECUTABLE}" tag --points-at HEAD WORKING_DIRECTORY "${JD_REPO}"
                  OUTPUT_VARIABLE JD_GIT_TAGS ERROR_QUIET)
  string(REGEX MATCH "(^|\n)v${PROJECT_VERSION}(\n|$)" JD_AT_TAG "${JD_GIT_TAGS}")
  if(JD_AT_TAG AND NOT JD_GIT_CHANGES)
    set(JD_VERSION_STRING "${PROJECT_VERSION}")
  elseif(NOT JD_GIT_FAILED AND JD_GIT_COMMIT)
    set(JD_VERSION_STRING "${JD_VERSION_STRING}+${JD_GIT_COMMIT}")
    if(JD_GIT_CHANGES)
      set(JD_VERSION_STRING "${JD_VERSION_STRING}.dirty")
    endif()
  endif()
endif()
message(STATUS "jackdaw ${JD_VERSION_STRING}")
# the day the build was configured, shown with the version in the sidebar

# in a source of their own, not compile definitions: those changed every file's flags with each commit and day,

file(CONFIGURE OUTPUT ${JD_GEN}/version.c

set_property(GLOBAL PROPERTY JD_VERSION_STRING "${JD_VERSION_STRING}")

# libetpan, static, as platform/*/deps.sh builds it. Linux takes OpenSSL, zlib and libcurl from the system;
# Android has none to link against, so platform/android/deps.sh builds them next to libetpan (JD_ANDROID_DEPS)
if(ANDROID)
  set(JD_ANDROID_DEPS "" CACHE PATH "OpenSSL, libcurl, libiconv and libetpan for this ABI (platform/android/deps.sh)")
  set(JD_LIBETPAN_DIR ${JD_ANDROID_DEPS})
else()
  set(JD_LIBETPAN_DIR "" CACHE PATH "libetpan's install prefix (include/libetpan, lib/libetpan.a)")
endif()
if(NOT EXISTS "${JD_LIBETPAN_DIR}/lib/libetpan.a")
  message(FATAL_ERROR "JD_LIBETPAN_DIR: no lib/libetpan.a in '${JD_LIBETPAN_DIR}' (platform/*/deps.sh builds it)")
endif()
if(ANDROID)
  foreach(lib ssl crypto curl iconv)
    add_library(jd_${lib} STATIC IMPORTED GLOBAL)
    set_target_properties(jd_${lib} PROPERTIES IMPORTED_LOCATION ${JD_ANDROID_DEPS}/lib/lib${lib}.a
                          INTERFACE_INCLUDE_DIRECTORIES ${JD_ANDROID_DEPS}/include)
  endforeach()
  set_target_properties(jd_ssl PROPERTIES INTERFACE_LINK_LIBRARIES jd_crypto)
  set_target_properties(jd_curl PROPERTIES INTERFACE_LINK_LIBRARIES "jd_ssl;jd_crypto;z"
                        INTERFACE_COMPILE_DEFINITIONS CURL_STATICLIB)
  set(JD_SSL_LIBS "jd_ssl;jd_crypto;z;jd_iconv;Threads::Threads")
  set(JD_CURL_LIB jd_curl)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(JD_SSL REQUIRED IMPORTED_TARGET openssl)
  pkg_check_modules(JD_ZLIB REQUIRED IMPORTED_TARGET zlib)
  pkg_check_modules(JD_CURL REQUIRED IMPORTED_TARGET libcurl)
  set(JD_SSL_LIBS "PkgConfig::JD_SSL;PkgConfig::JD_ZLIB;Threads::Threads")
  set(JD_CURL_LIB PkgConfig::JD_CURL)
endif()
add_library(jd_etpan STATIC IMPORTED GLOBAL)
set_target_properties(jd_etpan PROPERTIES IMPORTED_LOCATION ${JD_LIBETPAN_DIR}/lib/libetpan.a
                      INTERFACE_INCLUDE_DIRECTORIES ${JD_LIBETPAN_DIR}/include
                      INTERFACE_LINK_LIBRARIES "${JD_SSL_LIBS}")

  enable_testing()

  target_compile_options(jackdaw_keys_test PRIVATE ${JD_WARNINGS})
  target_link_libraries(jackdaw_keys_test PRIVATE jdapp)

  set_tests_properties(keys PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 60 ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0")
  # calendar data: zones, repeats, invitations
  add_executable(jackdaw_ical_test tests/cal/ical_test.c)

  target_link_libraries(jackdaw_ical_test PRIVATE jdcore)
